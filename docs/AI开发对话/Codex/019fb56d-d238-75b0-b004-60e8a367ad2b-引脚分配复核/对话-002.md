import fitz
from pathlib import Path
items=[
('deepseek', Path(r'C:\Users\16484\Desktop\H_MSPM0G3507_PinMap_Final.pdf')),
('schematic', Path(r'D:\A嵌入式\资料库\3507\立创·天猛星MSPM0G3507开发板资料-241115\立创·天猛星MSPM0G3507开发板资料\02-【MSPM0G3507】开源硬件\立创·天猛星MSPM0G3507开发板原理图.pdf')),
('pinout', Path(r'D:\A嵌入式\资料库\3507\立创·天猛星MSPM0G3507开发板资料-241115\立创·天猛星MSPM0G3507开发板资料\02-【MSPM0G3507】开源硬件\天猛星MSPM0G3507开发板引脚图.pdf')),
]
outdir=Path(r'C:\Users\16484\Desktop\pinmap_check_extract')
outdir.mkdir(exist_ok=True)
for name,p in items:
    doc=fitz.open(str(p))
    text=[]
    for i,page in enumerate(doc):
        text.append(f'\n===== PAGE {i+1} =====\n')
        text.append(page.get_text('text'))
        # 渲染图片，方便视觉核对原理图/引脚图
        pix=page.get_pixmap(matrix=fitz.Matrix(2,2), alpha=False)
        pix.save(str(outdir / f'{name}_p{i+1}.png'))
    (outdir / f'{name}.txt').write_text('\n'.join(text), encoding='utf-8')
print(outdir)
for f in sorted(outdir.iterdir()):
    print(f.name, f.stat().st_size)
