44	    DL_GPIO_clearPins(GPIO_LEDS_PORT, GPIO_LEDS_USER_LED_2_PIN);

49	    while (1) {
50	        /*
51	         * Call togglePins API to flip the current value of LEDs 1-3. This
52	         * API causes the corresponding HW bits to be flipped by the GPIO HW
53	         * without need for additional R-M-W cycles by the processor.
54	         */
55	        delay_cycles(DELAY);
56	        DL_GPIO_togglePins(GPIO_LEDS_PORT,
57	            GPIO_LEDS_USER_LED_1_PIN | GPIO_LEDS_USER_LED_2_PIN |
