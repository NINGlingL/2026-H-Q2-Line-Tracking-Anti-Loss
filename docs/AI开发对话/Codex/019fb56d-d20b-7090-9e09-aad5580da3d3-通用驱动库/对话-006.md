19	/*==============================================================================

23	/** 显存: 8页 × 128列 = 1024字节 */
24	static uint8_t OLED_GRAM[SSD1306_PAGE_COUNT][SSD1306_COLUMN];
25	
26	/** I2C 句柄和设备地址 */
27	static I2C_HandleTypeDef *oled_i2c   = NULL;
28	static uint8_t            oled_addr  = 0x78;
29	
30	/*==============================================================================
31	 * 底层 I2C 通信
32	 *============================================================================*/
33	
34	/**
35	 * @brief 向 OLED 发送数据
36	 * @param data 数据缓冲区
37	 * @param len  数据长度
38	 *
39	 * 控制字节已在数据首字节中:
40	 * - 0x00 + cmd  → 发送命令
41	 * - 0x40 + data → 发送显存数据
42	 */
43	static void OLED_Send(uint8_t *data, uint8_t len)
44	{
45	    if (oled_i2c == NULL) return;
46	    HAL_I2C_Master_Transmit(oled_i2c, (uint16_t)oled_addr, data, len,
47	                            HAL_MAX_DELAY);
48	}
49	
50	/** 发送单条命令 */
51	static void OLED_SendCmd(uint8_t cmd)
52	{
53	    uint8_t buf[2];
54	    buf[0] = 0x00;  /* 控制字节: 命令 */
55	    buf[1] = cmd;
56	    OLED_Send(buf, 2);
57	}

59	/*==============================================================================
60	 * SSD1306 初始化序列
61	 *============================================================================*/
62	
63	void OLED_Init(I2C_HandleTypeDef *hi2c, uint8_t i2c_addr)
64	{
65	    oled_i2c  = hi2c;
66	    oled_addr = (uint8_t)(i2c_addr << 0); /* 保持原始地址 */
67	
68	    /* ---- SSD1306 初始化命令序列 ---- */
69	    OLED_SendCmd(0xAE); /* 关闭显示 */
70	
71	    /* 设置内存寻址模式: 页寻址 */
72	    OLED_SendCmd(0x20);
73	    OLED_SendCmd(0x10);
74	
75	    /* 设置页起始地址 */
76	    OLED_SendCmd(0xB0);
77	
78	    /* COM 输出扫描方向: 从 COM[N-1] 到 COM0 */
79	    OLED_SendCmd(0xC8);
80	
81	    /* 设置列地址低4位 = 0 */
82	    OLED_SendCmd(0x00);
83	    /* 设置列地址高4位 = 0 */
84	    OLED_SendCmd(0x10);
85	
86	    /* 设置显示起始行 = 0 */
87	    OLED_SendCmd(0x40);
88	
89	    /* 对比度 */
90	    OLED_SendCmd(0x81);
91	    OLED_SendCmd(0xDF);
92	
93	    /* 段重映射: 列127映射到SEG0 */
94	    OLED_SendCmd(0xA1);
95	
96	    /* 正常显示 (非反色) */
97	    OLED_SendCmd(0xA6);
98	
99	    /* 多路复用比率: 64 */
100	    OLED_SendCmd(0xA8);
101	    OLED_SendCmd(0x3F);
102	
103	    /* 整个显示恢复为 RAM 内容 */
104	    OLED_SendCmd(0xA4);
105	
106	    /* 显示偏移 = 0 */
107	    OLED_SendCmd(0xD3);
108	    OLED_SendCmd(0x00);
109	
110	    /* 时钟分频/振荡器频率 */
111	    OLED_SendCmd(0xD5);
112	    OLED_SendCmd(0xF0);
113	
114	    /* 预充电周期 */
115	    OLED_SendCmd(0xD9);
116	    OLED_SendCmd(0x22);
117	
118	    /* COM 引脚配置 */
119	    OLED_SendCmd(0xDA);
120	    OLED_SendCmd(0x12);
121	
122	    /* VCOMH 电压 */
123	    OLED_SendCmd(0xDB);
124	    OLED_SendCmd(0x20);
125	
126	    /* 电荷泵设置 (DC-DC) */
127	    OLED_SendCmd(0x8D);
128	    OLED_SendCmd(0x14);
129	
130	    /* 清屏 */
131	    OLED_NewFrame();
132	    OLED_ShowFrame();
133	
134	    /* 开启显示 */
135	    OLED_SendCmd(0xAF);
136	}
