22	float PID_Compute(PID *pid, float sp, float mv, float dt)
23	{
24	    /* 首次调用: 不计算, 只初始化历史值 */
25	    if (pid->is_first) {
26	        pid->prev_measurement = mv;
27	        pid->prev_error       = sp - mv;
28	        pid->is_first         = false;
29	        return 0.0f;
30	    }
31	
32	    /* 误差 */
33	    float error = sp - mv;
34	
35	    /* ---- 比例项 ---- */
36	    float p_term = pid->Kp * error;
37	
38	    /* ---- 积分项 (带分离: 误差过大时清零积分, 防止 windup) ---- */
39	    if (fabsf(error) > pid->integral_max * 0.5f) {
40	        /* 大误差: 清零积分, 纯 P/D 控制 */
41	        pid->integral = 0.0f;
42	    } else {
43	        pid->integral += pid->Ki * error * dt;
44	        /* 积分限幅 */
45	        if (pid->integral >  pid->integral_max) pid->integral =  pid->integral_max;
46	        if (pid->integral < -pid->integral_max) pid->integral = -pid->integral_max;
47	    }
48	    float i_term = pid->integral;
49	
50	    /* ---- 微分项 (微分 on measurement, 避免微分冲击) ---- */
51	    float d_term = 0.0f;
52	    if (dt > 1e-6f) {
53	        d_term = -pid->Kd * (mv - pid->prev_measurement) / dt;
54	    }
55	
56	    /* ---- 合成输出 ---- */
57	    float out = p_term + i_term + d_term;
58	
59	    /* 输出限幅 */
60	    if (out >  pid->out_max) out =  pid->out_max;
61	    if (out < -pid->out_max) out = -pid->out_max;
62	
63	    /* 保存历史 */
64	    pid->prev_error       = error;
65	    pid->prev_measurement = mv;
66	
67	    /* NaN/Inf 保护 */
68	    if (!isfinite(out)) {
69	        PID_Reset(pid);
70	        return 0.0f;
71	    }
72	
73	    return out;
74	}
