    print(f'=== MSPM0 G3507 状态 ===')
    print(f'Probe: {s.probe.product_name}')
    print(f'UID:   {s.probe.unique_id}')
    print(f'Core:  Cortex-M0+ (CPUID 0x{t.read32(0xE000ED00):08X})')
    state = {T.Target.State.RUNNING:'RUNNING', T.Target.State.HALTED:'HALTED',
             T.Target.State.SLEEPING:'SLEEPING'}.get(t.get_state(),'?')
    print(f'State: {state}')
    if t.get_state() == T.Target.State.HALTED:
        print(f'PC:    0x{t.read_core_register(\"pc\"):08X}')
        print(f'SP:    0x{t.read_core_register(\"sp\"):08X}')
        print(f'LR:    0x{t.read_core_register(\"lr\"):08X}')
    print(f'Flash: 128KB @ 0x00000000')
    print(f'SRAM:   32KB @ 0x20000000')
    print(f'Clock: 32MHz SYSOSC')
