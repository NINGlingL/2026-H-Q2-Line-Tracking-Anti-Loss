description: Read MSPM0 chip info via pyocd
command: python -c "
from pyocd.core.helpers import ConnectHelper
from pyocd.core.target import Target
import time

with ConnectHelper.session_with_chosen_probe(blocking=False) as session:
    board = session.board
    target = session.target
    
    print('=== Chip Info ===')
    print(f'Target:   {target.part_number}')
    print(f'Vendor:   {target.vendor}')
    print(f'CPU:      Cortex-{target.cpu_type.name}')
    
    # Read CPUID
    cpuid = target.read32(0xE000ED00)
    print(f'CPUID:    0x{cpuid:08X}')
    
    # Check if halted
    if target.get_state() == Target.State.HALTED:
        print('State:    HALTED')
        pc = target.read_core_register('pc')
        print(f'PC:       0x{pc:08X}')
    else:
        print('State:    RUNNING')
    
    # Memory info
    flash = target.get_memory_map().get_region_for_address(0x00000000)
    if flash:
        print(f'Flash:    {flash.start:#010x} - {flash.end:#010x} ({flash.length//1024}KB)')
    ram = target.get_memory_map().get_region_for_address(0x20000000)
    if ram:
        print(f'RAM:      {ram.start:#010x} - {ram.end:#010x} ({ram.length//1024}KB)')
    
    print(f'Probe:    {session.probe.unique_id}')
" 2>&1
