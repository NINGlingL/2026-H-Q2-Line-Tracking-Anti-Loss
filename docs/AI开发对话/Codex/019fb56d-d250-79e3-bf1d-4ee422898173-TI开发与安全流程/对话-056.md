9	static volatile int32_t g_enc_sw_count = 0;
10	static volatile uint8_t  g_enc_sw_old  = 0;   /* bit1=prevA, bit0=prevB */
11	
12	/*
13	 *  正交解码 (4x 倍频, XOR 法)
14	 *  每次 A 或 B 跳变 → 比较 A^B → 确定方向
15	 */
16	static void sw_encoder_isr(void)
17	{
18	    uint32_t pins = DL_GPIO_readPins(ENC_SW_PORT, ENC_SW_A_PIN | ENC_SW_B_PIN);
19	    uint8_t  a    = (pins & ENC_SW_A_PIN) ? 1 : 0;
20	    uint8_t  b    = (pins & ENC_SW_B_PIN) ? 1 : 0;
21	
22	    uint8_t old_a = (g_enc_sw_old >> 1) & 1;
23	    if (a ^ old_a) {           /* A 相变化 */
24	        g_enc_sw_count += (a == b) ? 1 : -1;
25	    } else {                   /* B 相变化 */
26	        g_enc_sw_count += (a != b) ? 1 : -1;
27	    }
28	    g_enc_sw_old = (a << 1) | b;
29	}
