[external_agent_tool_call: Bash]
description: Read chip status after flash
command: python -c "
from pyocd.core.helpers import ConnectHelper
import pyocd.core.target as T
with ConnectHelper.session_with_chosen_probe(blocking=False) as s:
    t = s.target
    st = {T.Target.State.RUNNING:'RUNNING', T.Target.State.HALTED:'HALTED'}.get(t.get_state(),'?')
    print(f'Core:  Cortex-M0+  (CPUID 0x{t.read32(0xE000ED00):08X})')
    print(f'State: {st}')
    if t.get_state() == T.Target.State.HALTED:
        for r in ['pc','sp','lr']: print(f'{r.upper()}:    0x{t.read_core_register(r):08X}')
    print(f'FW:    ICM20948 IMU Display')
" 2>&1
[/external_agent_tool_call]
