# Print sorted by designator
import re
def sort_key(k):
    # Sort: letters first, then number
    m = re.match(r'([A-Z]+)(\d+)', k)
    if m:
        return (m.group(1), int(m.group(2)))
    return (k, 0)

for d in sorted(bom.keys(), key=sort_key):
    info = bom[d]
    # Determine what to show
    if info['description']:
        desc = info['description'].split(';')[0] if ';' in info['description'] else info['description'][:80]

    mf = info['manufacturer_part']
