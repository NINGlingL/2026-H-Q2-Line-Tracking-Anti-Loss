11	static volatile uint16_t g_wr = 0;    /* 写指针 (ISR 独占) */
12	static volatile uint16_t g_rd = 0;    /* 读指针 (主循环独占) */
13	static volatile uint32_t g_total = 0; /* 总接收字节数 */

15	/* ========== UART ISR ========== */

17	void UART2_IRQHandler(void)
18	{
19	    if (DL_UART_Main_getEnabledInterruptStatus(UART_0_INST,
20	            DL_UART_MAIN_INTERRUPT_RX) & DL_UART_MAIN_INTERRUPT_RX) {
21	        while (!DL_UART_Main_isRXFIFOEmpty(UART_0_INST)) {
22	            uint8_t c = DL_UART_Main_receiveData(UART_0_INST);
23	            g_total++;
24	            uint16_t next = (g_wr + 1) % UART_BT_BUF_SIZE;
25	            if (next != g_rd) {
26	                g_ring[g_wr] = c;
27	                g_wr = next;
28	            }
29	        }
30	        DL_UART_Main_clearInterruptStatus(UART_0_INST,
31	            DL_UART_MAIN_INTERRUPT_RX);
32	    }
33	}

43	{
44	    DL_UART_Main_disable(UART_0_INST);
45	    DL_UART_Main_configBaudRate(UART_0_INST, CPUCLK_FREQ, baud);
46	    DL_UART_Main_setRXFIFOThreshold(UART_0_INST,
47	        DL_UART_MAIN_RX_FIFO_LEVEL_ONE_ENTRY);

49	        DL_UART_MAIN_INTERRUPT_RX);

51	    DL_UART_Main_clearInterruptStatus(UART_0_INST, 0xFF);

58	uint16_t UartBT_Available(void)
59	{
60	    if (g_wr >= g_rd)
61	        return g_wr - g_rd;
62	    else
63	        return UART_BT_BUF_SIZE - g_rd + g_wr;
64	}
65	
66	bool UartBT_Read(uint8_t *c)
67	{
68	    if (g_rd == g_wr) return false;
69	    *c = g_ring[g_rd];
70	    g_rd = (g_rd + 1) % UART_BT_BUF_SIZE;
71	    return true;
72	}
73	
74	uint8_t UartBT_ReadBlocking(void)
75	{
76	    uint8_t c;
77	    while (!UartBT_Read(&c));
78	    return c;
79	}
80	
81	uint16_t UartBT_ReadLine(char *buf, uint16_t maxlen)
82	{
83	    uint16_t i;
84	    for (i = 0; i < maxlen - 1; i++) {
85	        uint8_t c;
86	        if (!UartBT_Read(&c)) break;
87	        if (c == '\n') { buf[i] = '\0'; return i; }
88	        if (c == '\r') continue;
89	        buf[i] = (char)c;
90	    }
91	    buf[i] = '\0';
92	    return i;
93	}
94	
95	uint16_t UartBT_ReadAll(uint8_t *buf, uint16_t maxlen)
96	{
97	    uint16_t i;
98	    for (i = 0; i < maxlen; i++) {
99	        if (!UartBT_Read(&buf[i])) break;
100	    }
101	    return i;
102	}

104	bool UartBT_Peek(uint16_t idx, uint8_t *c)
105	{
106	    if (idx >= UartBT_Available()) return false;
107	    *c = g_ring[(g_rd + idx) % UART_BT_BUF_SIZE];
108	    return true;
109	}
110	
111	void UartBT_Flush(uint16_t n)
112	{
113	    uint16_t avail = UartBT_Available();
114	    if (n > avail) n = avail;
115	    g_rd = (g_rd + n) % UART_BT_BUF_SIZE;
116	}
117	
118	void UartBT_Clear(void)
119	{
120	    g_rd = g_wr;
121	}

123	uint32_t UartBT_GetRxCount(void)
