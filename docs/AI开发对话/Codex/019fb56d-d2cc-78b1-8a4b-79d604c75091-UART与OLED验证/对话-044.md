12	static bool is_cmd(uint8_t c)
13	{
14	    if (c == 'F' || c == 'f') return true;
15	    if (c == 'B' || c == 'b') return true;
16	    if (c == 'L' || c == 'l') return true;
17	    if (c == 'R' || c == 'r') return true;
18	    if (c == 'S' || c == 's') return true;
19	    if (c == '+' || c == '-' || c == '=' || c == '_') return true;
20	    if (c >= '0' && c <= '9') return true;
21	    return false;
22	}
23	
24	static bool cmd_read(uint8_t *c)
25	{
26	    while (UartBT_Read(c)) {
27	        if (is_cmd(*c)) return true;
28	    }
29	    return false;
30	}

34	static void cmd_exec(char c)
35	{
36	    uint8_t sp = Moto_GetSpeed();
37	
38	    switch (c) {
39	    /* ---- 方向 ---- */
40	    case 'F': case 'f':
41	        Moto_Forward(sp ? sp : 50);
42	        snprintf(g_last, sizeof(g_last), "FWD  %u%%", Moto_GetSpeed());
43	        break;
44	    case 'B': case 'b':
45	        Moto_Backward(sp ? sp : 50);
46	        snprintf(g_last, sizeof(g_last), "BACK %u%%", Moto_GetSpeed());
47	        break;
48	    case 'L': case 'l':
49	        Moto_Left(sp ? sp : 50);
50	        snprintf(g_last, sizeof(g_last), "LEFT %u%%", Moto_GetSpeed());
51	        break;
52	    case 'R': case 'r':
53	        Moto_Right(sp ? sp : 50);
54	        snprintf(g_last, sizeof(g_last), "RIGHT %u%%", Moto_GetSpeed());
55	        break;
56	    case 'S': case 's':
57	        Moto_Stop();
58	        snprintf(g_last, sizeof(g_last), "STOP");
59	        break;
60	
61	    /* ---- 加减速 ---- */
62	    case '-': case '_':
63	        Moto_SpeedUp(10);
64	        snprintf(g_last, sizeof(g_last), "Spd+ %u%%", Moto_GetSpeed());
65	        break;
66	    case '+': case '=':
67	        Moto_SpeedDown(10);
68	        snprintf(g_last, sizeof(g_last), "Spd- %u%%", Moto_GetSpeed());
69	        break;
70	
71	    /* ---- 档位: 只调速, 方向为S时才切Forward ---- */
72	    case '0':
73	        Moto_Stop();
74	        snprintf(g_last, sizeof(g_last), "Spd 0%% STOP");
75	        break;
76	    case '1': case '2': case '3':
77	    case '4': case '5': case '6':
78	    case '7': case '8': case '9':
79	    {
80	        uint8_t pct = (uint8_t)(c - '0') * 10;
81	        Moto_SetSpeed(pct);
82	        snprintf(g_last, sizeof(g_last), "Spd %u%%", pct);
83	        if (Moto_GetDirection() == 'S') Moto_Forward(pct);
84	        break;
85	    }
86
