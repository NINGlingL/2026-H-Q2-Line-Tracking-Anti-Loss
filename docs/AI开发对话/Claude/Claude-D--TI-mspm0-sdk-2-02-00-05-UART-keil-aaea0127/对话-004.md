```c
/* 1. 系统初始化 */
SYSCFG_DL_init();

/* 1.5 给 PA11 (UART1 RX) 加上拉, 防止浮空 → 噪声 → ISR风暴 */
DL_GPIO_initPeripheralInputFunctionFeatures(
    GPIO_UART_1_IOMUX_RX,
```
